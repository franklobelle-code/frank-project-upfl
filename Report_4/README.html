medals_full <- medals %>%
  left_join(country, by = c("athlete_id", "Games")) %>%
  left_join(games, by = "Games")

top_15_nocs <- medals_full %>%
  filter(!is.na(Medal)) %>%
  count(NOC, sort = TRUE) %>%
  head(15) 

medal_matrix <- medals_full %>%
  filter(!is.na(Medal)) %>%
  inner_join(top_15_nocs, by = "NOC") %>%
  count(NOC, Year, name = "Total_Medals") %>%
  pivot_wider(names_from = Year, values_from = Total_Medals, values_fill = 0) %>%
  # column with total medals
  mutate(num_cols = select(., where(is.numeric)),
         Total = rowSums(num_cols)) %>% 
  select(-num_cols)
  

medal_matrix
medal_success <- medals_full %>%
  inner_join(top_15_nocs, by = "NOC") %>%
  mutate(won_medal = if_else(!is.na(Medal), "Won", "No_Medal"))

success_rate <- medal_success %>%
  group_by(NOC) %>%
  summarise(total_competed = n(),
    total_won = sum(!is.na(Medal)),
    success_pct = (total_won / total_competed) * 100)

success_rate

The chart shows that medal chances change by country

URS and GDR show the highest success rates, reaching around 40%, meaning they earned medals in a large share of the events they entered.

USA is next at about 30%, followed by RUS and GER in the low 20s.

FRA and CAN are in the lower range, with success rates around 13–15%.

The Chi Square test

crosstab_table <- table(medal_success$NOC, medal_success$won_medal)
chi_result <- chisq.test(crosstab_table)
chi_result
## 
##  Pearson's Chi-squared test
## 
## data:  crosstab_table
## X-squared = 4531.9, df = 14, p-value < 2.2e-16

This result confirms what the bar chart already suggested: Countries like URS and GDR reach rates near 40%, while others stay around 13–15%. The data shows that success is not only about how many events a country enters, but that medal probability varies clearly from one country to another

Part 6.

sport_data <- medals_full %>%
  left_join(athletes, by = c("athlete_id" = "ID")) %>%
  group_by(Sport) %>%
  summarise(height = mean(Height, na.rm = TRUE),
    weight = mean(Weight, na.rm = TRUE)) %>%
  mutate(bmi = weight / ((height / 100)^2)) %>%
  filter(!is.na(height) & !is.na(weight))

extremes <- sport_data %>%
  mutate(label = case_when(
      height == max(height, na.rm = TRUE) ~ paste0(Sport, ": Max Height"),
      height == min(height, na.rm = TRUE) ~ paste0(Sport, ": Min Height"),
      weight == max(weight, na.rm = TRUE) ~ paste0(Sport, ": Max Weight"),
      weight == min(weight, na.rm = TRUE) ~ paste0(Sport, ": Min Weight"),
      bmi == max(bmi, na.rm = TRUE) ~ paste0(Sport, ": Max BMI"),
      bmi == min(bmi, na.rm = TRUE) ~ paste0(Sport, ": Min BMI"),
      TRUE ~ NA_character_)) %>%
  filter(!is.na(label))

# Plot
sport_data %>%
  ggplot(aes(x = weight, y = height)) +
  geom_point(color = "gray", alpha = 0.7) +
  geom_point(data = extremes, color = "red", size = 3) +
